local CurrentTheme = Themes[Settings.Theme]
local ThemeObjects = { MainBg = {}, HeaderBg = {}, CardBg = {}, Accent = {}, Text = {}, SubText = {}, Border = {} }

local function TrackThemeElement(element, property, category)
    table.insert(ThemeObjects[category], {Object = element, Property = property})
    element[property] = CurrentTheme[category]
end

local function SwitchTheme(themeName)
    if not Themes[themeName] then return end
    Settings.Theme = themeName
    CurrentTheme = Themes[themeName]
    for category, items in pairs(ThemeObjects) do
        for _, item in ipairs(items) do
            if item.Object and item.Object.Parent then
                item.Object[item.Property] = CurrentTheme[category]
            end
        end
    end
end

-- buttonanim
local function AddButtonEffects(button)
    local gradient = Instance.new('UIGradient')
    gradient.Rotation = 90
    gradient.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(0.40, Color3.fromRGB(230, 230, 235)),
        ColorSequenceKeypoint.new(1.00, Color3.fromRGB(160, 160, 175))
    }
    gradient.Parent = button

    local tweenInfo = TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

    button.MouseButton1Down:Connect(function()
        local originalSize = button.Size
        button:SetAttribute("OrigSize", originalSize)
        TweenService:Create(button, tweenInfo, {
            Size = UDim2.new(originalSize.X.Scale * 0.96, originalSize.X.Offset * 0.96, originalSize.Y.Scale * 0.96, originalSize.Y.Offset * 0.96)
        }):Play()
    end)
    
    local function restoreSize()
        local originalSize = button:GetAttribute("OrigSize")
        if originalSize then
            TweenService:Create(button, tweenInfo, {Size = originalSize}):Play()
        end
    end

    button.MouseButton1Up:Connect(restoreSize)
    button.MouseLeave:Connect(restoreSize)
end

-- config file saving and loading
local function SaveConfig()
    local encodedData = HttpService:JSONEncode(Settings)
    if writefile then
        pcall(function() writefile("CakepopHitbox_Config.json", encodedData) end)
    end
end

local function LoadConfig()
    local rawData = nil
    if readfile and isfile and isfile("CakepopHitbox_Config.json") then
        pcall(function() rawData = readfile("CakepopHitbox_Config.json") end)
    end
    
    if rawData then
        local success, decoded = pcall(function() return HttpService:JSONDecode(rawData) end)
        if success and type(decoded) == "table" then
            for key, value in pairs(decoded) do Settings[key] = value end
            if Settings.Theme and Themes[Settings.Theme] then SwitchTheme(Settings.Theme) end
            Lighting.FogStart = Settings.FogStart
            Lighting.FogEnd = Settings.FogEnd
            Lighting.FogColor = Color3.fromRGB(unpack(Settings.FogColor))
            Lighting.Ambient = Color3.fromRGB(unpack(Settings.Ambient))
            Lighting.OutdoorAmbient = Color3.fromRGB(unpack(Settings.OutdoorAmbient))
        end
    end
end

-- autoloadcheck
local startupConfig = nil
if readfile and isfile and isfile("CakepopHitbox_Config.json") then
    pcall(function() startupConfig = readfile("CakepopHitbox_Config.json") end)
end
if startupConfig then
    pcall(function()
        local data = HttpService:JSONDecode(startupConfig)
        if data and data.AutoLoad then LoadConfig() end
    end)
end

local isDraggingWindow, dragInputStart, windowStartPos
HeaderBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isDraggingWindow = true
        dragInputStart = input.Position
        windowStartPos = MainWindow.Position
        
        local connection
        connection = input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                isDraggingWindow = false
                connection:Disconnect()
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isDraggingWindow and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragInputStart
        MainWindow.Position = UDim2.new(windowStartPos.X.Scale, windowStartPos.X.Offset + delta.X, windowStartPos.Y.Scale, windowStartPos.Y.Offset + delta.Y)
    end
end)

-- wintoggle
local isUIVisible = true
local isTweeningWindow = false
local normalPosition = UDim2.new(0.5, -220, 0.25, 0)
local hiddenPosition = UDim2.new(0.5, -220, 0.25, 20)

local function ToggleWindowVisibility(show)
    if isTweeningWindow or isUIVisible == show then return end
    isTweeningWindow = true

    if show then
        MainWindow.Visible = true
        MainWindow.Size = UDim2.new(0, 440 * 0.85, 0, 380 * 0.85)
        MainWindow.Position = hiddenPosition
        MainWindow.BackgroundTransparency = 1
        ContainerGroup.GroupTransparency = 1
        
        TweenService:Create(MainWindow, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = 0 }):Play()
        TweenService:Create(ContainerGroup, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { GroupTransparency = 0 }):Play()

        local popIn = TweenService:Create(MainWindow, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 440, 0, 380),
            Position = normalPosition
        })
        popIn:Play()
        popIn.Completed:Wait()
        isUIVisible = true
    else
        TweenService:Create(MainWindow, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { BackgroundTransparency = 1 }):Play()
        TweenService:Create(ContainerGroup, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { GroupTransparency = 1 }):Play()

        local popOut = TweenService:Create(MainWindow, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 440 * 0.85, 0, 380 * 0.85),
            Position = hiddenPosition
        })
        popOut:Play()
        popOut.Completed:Wait()
        MainWindow.Visible = false
        isUIVisible = false
    end

    isTweeningWindow = false
end

local ActiveSliders = {}

local function BuildSlider(parent, labelText, minVal, maxVal, defaultVal, callback)
    local Frame = Instance.new('Frame')
    Frame.Size = UDim2.new(1, 0, 0, 42)
    Frame.BorderSizePixel = 0
    Frame.Parent = parent
    TrackThemeElement(Frame, "BackgroundColor3", "CardBg")

    local Corner = Instance.new('UICorner')
    Corner.CornerRadius = UDim.new(0, 8)
    Corner.Parent = Frame

    local Stroke = Instance.new('UIStroke')
    Stroke.Thickness = 0.8
    Stroke.Parent = Frame
    TrackThemeElement(Stroke, "Color", "Border")

    local Label = Instance.new('TextLabel')
    Label.Size = UDim2.new(0.5, 0, 0, 18)
    Label.Position = UDim2.new(0, 12, 0, 4)
    Label.BackgroundTransparency = 1
    Label.Text = labelText
    Label.Font = Enum.Font.Gotham
    Label.TextSize = 10
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Frame
    TrackThemeElement(Label, "TextColor3", "Text")

    local ValBtn = Instance.new('TextButton')
    ValBtn.Size = UDim2.new(0, 50, 0, 18)
    ValBtn.Position = UDim2.new(1, -62, 0, 4)
    ValBtn.BackgroundTransparency = 1
    ValBtn.Text = ""
    ValBtn.Parent = Frame

    local ValBox = Instance.new('TextBox')
    ValBox.Size = UDim2.new(1, 0, 1, 0)
    ValBox.BackgroundTransparency = 1
    ValBox.Text = tostring(defaultVal)
    ValBox.Font = Enum.Font.GothamBold
    ValBox.TextSize = 10
    ValBox.TextXAlignment = Enum.TextXAlignment.Right
    ValBox.Parent = ValBtn
    TrackThemeElement(ValBox, "TextColor3", "SubText")

    local Track = Instance.new('Frame')
    Track.Size = UDim2.new(1, -24, 0, 4)
    Track.Position = UDim2.new(0, 12, 1, -12)
    Track.BorderSizePixel = 0
    Track.Parent = Frame
    TrackThemeElement(Track, "BackgroundColor3", "MainBg")

    local TrackCorner = Instance.new('UICorner')
    TrackCorner.CornerRadius = UDim.new(1, 0)
    TrackCorner.Parent = Track

    local FillBar = Instance.new('Frame')
    FillBar.Size = UDim2.new((defaultVal - minVal)/(maxVal - minVal), 0, 1, 0)
    FillBar.BorderSizePixel = 0
    FillBar.Parent = Track
    TrackThemeElement(FillBar, "BackgroundColor3", "Accent")

    local FillCorner = Instance.new('UICorner')
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = FillBar

    local minLimit, maxLimit = minVal, maxVal

    local function ApplySliderValue(value)
        value = math.floor(value * 100) / 100
        value = math.clamp(value, minLimit, maxLimit)
        local positionRatio = (value - minLimit) / (maxLimit - minLimit)
        FillBar.Size = UDim2.new(math.clamp(positionRatio, 0, 1), 0, 1, 0)
        ValBox.Text = tostring(value)
        callback(value)
        if Settings.Autosave then SaveConfig() end
    end

    local function ComputeInputPosition(input)
        local ratio = math.clamp((input.Position.X - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
        local calculatedValue = minLimit + (maxLimit - minLimit) * ratio
        ApplySliderValue(calculatedValue)
    end

    local isDraggingSlider = false
    Track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDraggingSlider = true
            ComputeInputPosition(input)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if isDraggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            ComputeInputPosition(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDraggingSlider = false
        end
    end)

    ValBox.FocusLost:Connect(function()
        local parsedNumber = tonumber(ValBox.Text)
        if parsedNumber then
            ApplySliderValue(parsedNumber)
        else
            ApplySliderValue(defaultVal)
        end
    end)

    local sliderController = {
        UpdateBounds = function(newMin, newMax)
            minLimit = newMin
            maxLimit = newMax
            local currentVal = tonumber(ValBox.Text) or defaultVal
            if currentVal > newMax then currentVal = newMax end
            ApplySliderValue(currentVal)
        end
    }

    if labelText == "Hitbox Size" then
        table.insert(ActiveSliders, sliderController)
    end

    return sliderController
end

-- Hotkey Listeners
if not IsMobile then
    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if isBindingKey and input.UserInputType == Enum.UserInputType.Keyboard then
            isBindingKey = false
            if input.KeyCode == Enum.KeyCode.RightBracket or input.KeyCode.LeftBracket or input.KeyCode.Backspace or input.KeyCode.Delete or input.KeyCode.Escape then
                Settings.ToggleKey = "None"
                KeybindButton.Text = "None"
            else
                Settings.ToggleKey = input.KeyCode.Name
                KeybindButton.Text = Settings.ToggleKey
            end
            if Settings.Autosave then SaveConfig() end
            return
        end

        if not gameProcessed and input.UserInputType == Enum.UserInputType.Keyboard then
            local pressedKeyName = input.KeyCode.Name
            if pressedKeyName == Settings.ToggleKey then
                ToggleWindowVisibility(not isUIVisible)
            elseif pressedKeyName == Settings.SpeedKey and Settings.RageMode then
                Settings.SpeedEnabled = not Settings.SpeedEnabled
                if SpeedToggle then SpeedToggle.SetState(Settings.SpeedEnabled) end
            elseif pressedKeyName == Settings.FlyKey and Settings.RageMode then
                Settings.FlyEnabled = not Settings.FlyEnabled
                if FlyToggle then FlyToggle.SetState(Settings.FlyEnabled) end
            end
        end
    end)
end

-- Core Game Loop (Hitbox expansion, flight mechanics, and performance tracking)
local SavedPartProperties = {}
local CachedTargetParts = {}
local CacheTimer = 0

local activeFlyVelocity, activeFlyGyro
local function CleanUpFlight()
    if activeFlyVelocity then activeFlyVelocity:Destroy() activeFlyVelocity = nil end
    if activeFlyGyro then activeFlyGyro:Destroy() activeFlyGyro = nil end
end

RunService.Heartbeat:Connect(function(deltaTime)
    local character = LocalPlayer.Character
    if character and character:FindFirstChild("Humanoid") then
        local humanoid = character.Humanoid
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        
        if Settings.RageMode and Settings.SpeedEnabled then
            humanoid.WalkSpeed = Settings.SpeedValue
        end

        if Settings.RageMode and Settings.FlyEnabled and rootPart then
            if not activeFlyVelocity then
                activeFlyVelocity = Instance.new("BodyVelocity")
                activeFlyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                activeFlyVelocity.Velocity = Vector3.new(0, 0, 0)
                activeFlyVelocity.Parent = rootPart

                activeFlyGyro = Instance.new("BodyGyro")
                activeFlyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
                activeFlyGyro.CFrame = rootPart.CFrame
                activeFlyGyro.Parent = rootPart
            end
            
            local cameraCF = Camera.CFrame
            local direction = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then direction = direction + cameraCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then direction = direction - cameraCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then direction = direction - cameraCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then direction = direction + cameraCF.RightVector end
            
            activeFlyVelocity.Velocity = direction * Settings.FlySpeed
            activeFlyGyro.CFrame = cameraCF
        else
            CleanUpFlight()
        end
    else
        CleanUpFlight()
    end

    -- Periodically update target cache to save system resources
    CacheTimer = CacheTimer + deltaTime
    if CacheTimer >= 1.0 then
        CacheTimer = 0
        table.clear(CachedTargetParts)

        if Settings.HitboxEnabled then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local targetRoot = player.Character:FindFirstChild("HumanoidRootPart") or player.Character:FindFirstChild("Torso")
                    if targetRoot and (not LocalPlayer.Character or targetRoot.Parent ~= LocalPlayer.Character) then 
                        table.insert(CachedTargetParts, targetRoot) 
                    end
                end
            end

            if Settings.TargetNPCs then
                for _, model in ipairs(workspace:GetChildren()) do
                    if model:IsA("Model") and model ~= LocalPlayer.Character and not Players:GetPlayerFromCharacter(model) then
                        if model:FindFirstChildOfClass("Humanoid") then
                            local targetRoot = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
                            if targetRoot then table.insert(CachedTargetParts, targetRoot) end
                        end
                    elseif model:IsA("Folder") then
                        for _, subModel in ipairs(model:GetChildren()) do
                            if subModel:IsA("Model") and subModel ~= LocalPlayer.Character and subModel:FindFirstChildOfClass("Humanoid") then
                                local targetRoot = subModel:FindFirstChild("HumanoidRootPart") or subModel:FindFirstChild("Torso")
                                if targetRoot then table.insert(CachedTargetParts, targetRoot) end
                            end
                        end
                    end
                end
            end
        end
    end

    if not Settings.HitboxEnabled then
        if next(SavedPartProperties) ~= nil then
            for part, original in pairs(SavedPartProperties) do
                if part and part.Parent then
                    part.Size = original.Size
                    part.Transparency = original.Transparency
                    part.CanCollide = original.CanCollide
                    part.Color = original.Color
                    part.Material = original.Material
                    local outlineBox = part:FindFirstChild("CakepopAdornment")
                    if outlineBox then outlineBox:Destroy() end
                end
            end
            table.clear(SavedPartProperties)
            table.clear(CachedTargetParts)
        end
        return
    end

    local expandedSize = Vector3.new(Settings.HitboxSize, Settings.HitboxSize, Settings.HitboxSize)
    local selectedHitboxColor = Color3.fromRGB(unpack(Settings.HitboxColor))
    local computedTransparency = 1 - Settings.HitboxOpacity
    local selectedOutlineColor = Color3.fromRGB(unpack(Settings.OutlineColor))
    local activePartsThisTick = {}

    for _, rootPart in ipairs(CachedTargetParts) do
        if rootPart and rootPart.Parent then
            activePartsThisTick[rootPart] = true
            
            if not SavedPartProperties[rootPart] then
                SavedPartProperties[rootPart] = {
                    Size = rootPart.Size,
                    Transparency = rootPart.Transparency,
                    CanCollide = rootPart.CanCollide,
                    Color = rootPart.Color,
                    Material = rootPart.Material
                }
            end

            if rootPart.Size ~= expandedSize then rootPart.Size = expandedSize end
            if rootPart.Transparency ~= computedTransparency then rootPart.Transparency = computedTransparency end
            if rootPart.Color ~= selectedHitboxColor then rootPart.Color = selectedHitboxColor end
            if rootPart.CanCollide ~= false then rootPart.CanCollide = false end
            if rootPart.Material ~= Enum.Material.SmoothPlastic then rootPart.Material = Enum.Material.SmoothPlastic end

            local outlineBox = rootPart:FindFirstChild("CakepopAdornment")
            if Settings.OutlineEnabled then
                if not outlineBox then
                    outlineBox = Instance.new("SelectionBox")
                    outlineBox.Name = "CakepopAdornment"
                    outlineBox.Adornee = rootPart
                    outlineBox.Parent = rootPart
                end
                if outlineBox.Color3 ~= selectedOutlineColor then outlineBox.Color3 = selectedOutlineColor end
                if outlineBox.LineThickness ~= Settings.OutlineThickness then outlineBox.LineThickness = Settings.OutlineThickness end
                if outlineBox.Transparency ~= (1 - Settings.OutlineOpacity) then outlineBox.Transparency = 1 - Settings.OutlineOpacity end
            else
                if outlineBox then outlineBox:Destroy() end
            end
        end
    end

    -- revert parts that are no longer targeted
    for part, original in pairs(SavedPartProperties) do
        if not activePartsThisTick[part] then
            if part and part.Parent then
                part.Size = original.Size
                part.Transparency = original.Transparency
                part.CanCollide = original.CanCollide
                part.Color = original.Color
                part.Material = original.Material
                local outlineBox = part:FindFirstChild("CakepopAdornment")
                if outlineBox then outlineBox:Destroy() end
            end
            SavedPartProperties[part] = nil
        end
    end
end)
